```C#
using Codice.Utils;
using System;
using System.IO;
using System.Threading.Tasks;
using UnityEngine;

namespace DevelopProducts.SaveSystem
{
    /// <summary>
    ///    セーブデータの基底クラス。
    /// </summary>
    public abstract class SaveBase
    {
        /// <summary>
        ///     セーブデータを非同期で読み込みます。 
        /// </summary>
        /// <returns></returns>
        internal async ValueTask ReadAsync()
        {
            if (!File.Exists(FilePath))
                return;

            var json = await File.ReadAllTextAsync(FilePath);
            JsonUtility.FromJsonOverwrite(json, this);
        }
        /// <summary>
        ///     セーブデータを非同期で保存します。
        /// </summary>
        /// <returns></returns>
        internal async ValueTask WriteAsync()
        {
            try
            {
                var json = JsonUtility.ToJson(this, true);
                var tempPath = FilePath + ".tmp";
                await File.WriteAllTextAsync(tempPath, json);

                if (File.Exists(FilePath))
                    File.Delete(FilePath);

                File.Move(tempPath, FilePath);
            }
            catch (Exception ex)
            {
                Debug.LogError($"Failed to write save file: {ex}");
                throw;
            }
        }
        /// <summary> セーブデータのキーを取得します。</summary>
        private string SaveKey => GetType().Name;
        /// <summary> セーブデータのファイルパスを取得します。</summary>
        private string FilePath => _filePath ??= Path.Combine(Application.persistentDataPath, $"{SaveKey}.json");
        private string _filePath;
    }
}
```

```C#
using System;
using System.Collections.Generic;
using System.Threading.Tasks;

namespace DevelopProducts.SaveSystem
{
    /// <summary>
    ///     セーブデータの管理システム。
    /// </summary>
    public static class SaveSystem
    {
        /// <summary>
        ///    指定された型のセーブデータを非同期で読み込みます。
        ///    キャッシュが存在する場合はキャッシュから返し、存在しない場合は新たに読み込みます。
        /// </summary>
        /// <typeparam name="T">セーブデータの型。</typeparam>
        /// <returns>指定された型のセーブデータ。</returns>
        public static ValueTask<T> LoadAsync<T>() where T : SaveBase, new()
        {
            var type = typeof(T);

            // ロックを取得して、キャッシュと読み込みタスクを確認します。
            lock (_lock)
            {
                // キャッシュに指定された型のデータが存在する場合は、キャッシュから返します。
                if (_cache.TryGetValue(type, out var cached))
                    return new((T)cached);

                // 読み込みタスクが存在する場合は、そのタスクを待機して結果を返します。
                if (_loadingTasks.TryGetValue(type, out var loadingTask))
                    return AwaitLoadingTask<T>(loadingTask.Value);

                // キャッシュにも読み込みタスクにも存在しない場合は、新たに読み込みタスクを作成して保存します。
                var lazyTask = new Lazy<Task<SaveBase>>(() => LoadInternalAsync<T>(type));
                _loadingTasks[type] = lazyTask;
                return AwaitAndCleanup<T>(type, lazyTask.Value);
            }
        }
        /// <summary>
        ///   指定されたセーブデータを非同期で保存します。
        /// </summary>
        /// <typeparam name="T">セーブデータの型。</typeparam>
        /// <param name="data">保存するセーブデータ。</param>
        /// <returns>非同期操作のタスク。</returns>
        public static async ValueTask SaveAsync<T>(T data) where T : SaveBase
        {
            await data.WriteAsync();
            // ロックを取得して、キャッシュに保存されたデータを更新します。
            lock (_lock)
            {
                _cache[typeof(T)] = data;
            }
        }
        /// <summary>
        ///   指定された型のセーブデータをキャッシュから削除します。
        /// </summary>
        /// <typeparam name="T">セーブデータの型。</typeparam>
        public static void Unload<T>() where T : SaveBase
        {
            var type = typeof(T);
            // ロックを取得して、キャッシュと読み込みタスクから指定された型のデータを削除します。
            lock (_lock)
            {
                _cache.Remove(type);
                _loadingTasks.Remove(type);
            }
        }
        /// <summary>
        ///   指定された型のセーブデータの読み込みタスクを待機し、
        ///   完了後にキャッシュからクリーンアップします。
        /// </summary>
        /// <typeparam name="T">セーブデータの型。</typeparam>
        /// <param name="type">セーブデータの型情報。</param>
        /// <param name="task">読み込みタスク。</param>
        /// <returns>指定された型のセーブデータ。</returns>
        private static async ValueTask<T> AwaitAndCleanup<T>(Type type, Task<SaveBase> task)
            where T : SaveBase
        {
            // タスクの完了を待機し、結果を取得します。
            try
            {
                return (T)await task;
            }
            // タスクの完了後に、ロックを取得して読み込みタスクをクリーンアップします。
            finally
            {
                lock (_lock)
                {
                    _loadingTasks.Remove(type);
                }
            }
        }
        /// <summary>
        ///    taskを待機して、完了後に指定された型のセーブデータを返します。
        /// </summary>
        /// <typeparam name="T">セーブデータの型。</typeparam>
        /// <param name="task">読み込みタスク。</param>
        /// <returns>指定された型のセーブデータ。</returns>
        private static async ValueTask<T> AwaitLoadingTask<T>(Task<SaveBase> task)
            where T : SaveBase
        {
            // タスクの完了を待機し、結果を取得します。
            return (T)await task;
        }
        /// <summary>
        ///  loadAsyncの内部実装。指定された型のセーブデータを非同期で読み込み、キャッシュに保存します。
        /// </summary>
        /// <typeparam name="T">セーブデータの型。</typeparam>
        /// <param name="type">セーブデータの型情報。</param>
        /// <returns>指定された型のセーブデータ。</returns>
        private static async Task<SaveBase> LoadInternalAsync<T>(Type type)
            where T : SaveBase, new()
        {
            var instance = new T();
            await instance.ReadAsync();
            lock (_lock)
            {
                _cache[type] = instance;
            }
            return instance;
        }
        /// <summary>キャッシュされたセーブデータを保持する辞書。</summary>
        private static readonly Dictionary<Type, SaveBase> _cache = new();
        /// <summary>読み込み中のセーブデータのタスクを保持する辞書。</summary>
        private static readonly Dictionary<Type, Lazy<Task<SaveBase>>> _loadingTasks = new();
        /// <summary>キャッシュと読み込みタスクへのアクセスを同期するためのロックオブジェクト。</summary>
        private static readonly object _lock = new();
    }
}
```

### 基本構造

### 概要
  本システムは「型ごとにセーブデータを管理する汎用セーブシステム」です。
  - ジェネリックで型安全に扱う
  - 非同期ロード・セーブ対応
  - キャッシュとロード中タスクの共有で重複処理を防止
---

### クラス構成

### SaveSystem（静的クラス）
  セーブデータの管理ハブ
  役割：
  - データのロード・セーブ
  - キャッシュ管理
  - 同時ロード制御
---

### SaveBase（抽象クラス）
  セーブデータの基底クラス
  役割：
  - JSONシリアライズ
  - ファイルI/O
  - 各データの共通処理

### データフロー

### LoadAsync の流れ
  1. キャッシュ確認
  2. 読み込み中タスク確認
  3. なければ新規ロード開始
  4. 読み込み完了後キャッシュ保存
  textLoadAsync<T>()
 ├─ cache に存在 → 即返す
 ├─ loadingTasks に存在 → await
 └─ 新規ロード
     ├─ ReadAsync()
     ├─ cache に保存
     └─ loadingTasks から削除
```C#
/// <summary>
///    指定された型のセーブデータを非同期で読み込みます。
///    キャッシュが存在する場合はキャッシュから返し、存在しない場合は新たに読み込みます。
/// </summary>
/// <typeparam name="T">セーブデータの型。</typeparam>
/// <returns>指定された型のセーブデータ。</returns>
public static ValueTask<T> LoadAsync<T>() where T : SaveBase, new()
{
    var type = typeof(T);

    // ロックを取得して、キャッシュと読み込みタスクを確認します。
    lock (_lock)
    {
        // キャッシュに指定された型のデータが存在する場合は、キャッシュから返します。
        if (_cache.TryGetValue(type, out var cached))
            return new((T)cached);

        // 読み込みタスクが存在する場合は、そのタスクを待機して結果を返します。
        if (_loadingTasks.TryGetValue(type, out var loadingTask))
            return AwaitLoadingTask<T>(loadingTask.Value);

        // キャッシュにも読み込みタスクにも存在しない場合は、新たに読み込みタスクを作成して保存します。
        var lazyTask = new Lazy<Task<SaveBase>>(() => LoadInternalAsync<T>(type));
        _loadingTasks[type] = lazyTask;
        return AwaitAndCleanup<T>(type, lazyTask.Value);
    }
}
/// <summary>
///   指定された型のセーブデータの読み込みタスクを待機し、
///   完了後にキャッシュからクリーンアップします。
/// </summary>
/// <typeparam name="T">セーブデータの型。</typeparam>
/// <param name="type">セーブデータの型情報。</param>
/// <param name="task">読み込みタスク。</param>
/// <returns>指定された型のセーブデータ。</returns>
private static async ValueTask<T> AwaitAndCleanup<T>(Type type, Task<SaveBase> task)
    where T : SaveBase
{
    // タスクの完了を待機し、結果を取得します。
    try
    {
        return (T)await task;
    }
    // タスクの完了後に、ロックを取得して読み込みタスクをクリーンアップします。
    finally
    {
        lock (_lock)
        {
            _loadingTasks.Remove(type);
        }
    }
}
/// <summary>
///    taskを待機して、完了後に指定された型のセーブデータを返します。
/// </summary>
/// <typeparam name="T">セーブデータの型。</typeparam>
/// <param name="task">読み込みタスク。</param>
/// <returns>指定された型のセーブデータ。</returns>
private static async ValueTask<T> AwaitLoadingTask<T>(Task<SaveBase> task)
    where T : SaveBase
{
    // タスクの完了を待機し、結果を取得します。
    return (T)await task;
}
/// <summary>
///  loadAsyncの内部実装。指定された型のセーブデータを非同期で読み込み、キャッシュに保存します。
/// </summary>
/// <typeparam name="T">セーブデータの型。</typeparam>
/// <param name="type">セーブデータの型情報。</param>
/// <returns>指定された型のセーブデータ。</returns>
private static async Task<SaveBase> LoadInternalAsync<T>(Type type)
    where T : SaveBase, new()
{
    var instance = new T();
    await instance.ReadAsync();
    lock (_lock)
    {
        _cache[type] = instance;
    }
    return instance;
}
```

### SaveAsync の流れ
  1. JSON化して保存
  2. キャッシュ更新
  textSaveAsync<T>()
 ├─ WriteAsync()
 └─ cache 更新
---

### キャッシュ設計
  `csharpDictionary<Type, SaveBase> _cache`
  - 型ごとに1インスタンスを保持
  - 同一データの多重ロードを防止
---

### 同時実行制御
  `csharpDictionary<Type, Lazy<Task<SaveBase>>> _loadingTasks`
  目的：
  - 同じデータの同時ロードを防ぐ
  - 1回のロード結果を全呼び出しで共有
---

### スレッド安全性
  `csharplock (_lock)`
  - キャッシュとタスク辞書を保護
  - 非同期＋マルチスレッド環境で安全

### MEMO

#### ValueTask
    [ValueTaskについての記事](https://www.hanachiru-blog.com/entry/2022/11/03/120000)
    非同期の結果をTaskよりも軽くできる仕組み。特に「すぐ完了することが多い処理」で、余計な割り当てを減らしたいときに使う

### 使いどころ
    - キャッシュ済みデータを返すAPI
    - 同期的に終わることが多いI/O処理
    - 高頻度で呼ばれる処理の最適化

#### Lazyクラス
    [Lazyクラスについて](https://qiita.com/sebayashi-tomoya/items/09d5825b0935ea372c73)
    
  今あるセーブシステムは恐らくクリーンアーキテクチャにのっとって作成されたものであり今回はセーブシステムごとUtility化したほうがベストだと考え、もとからあった機能をUtilityにまとめた。
  セーブシステムは他のモジュールに影響を及ぼすため、なるべくクライアント側でのセーブデータの作成コストを抑えることを第一目標にして制作した。
