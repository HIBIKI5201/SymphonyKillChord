#### UI変更
Adaptor層
```C#
public class Controller
{
		public void Update()
		{
				Domain.UpdateParameter();
				_presenter.Push();
		}
		
		private Presenter _presenter;
}

public class Presenter
{
		public void Push()
		{
				float changeValue = Domain.Get();
				DTO dto = new(changeValue);
				_vm.Push(dto);
		}
		
		private IViewModel _vm;
}

public interface IViewModel
{
		public void Push(DTO dto);
}

public readonly struct DTO
{
		public DTO(float value)
		{
				Value = value;
		}
		
		public readonly float Value;
}
```
UI層
```C#
public class ViewModel : IViewModel
{
		public void Push(DTO dto)
		{
				Value = dto.Value;
				OnChanged?.Invoke(Value);
		}
		
		public event Action<float> OnChanged;
		public float Value { get; private set; }
}

public class View
{
		public void Bind(ViewModel vm)
		{
				vm.OnChanged += OnChangedHandler;
		}
		
		private void OnChangedHandler(float value)
		{
				_text.text = value.ToString();
		}
		
		private Text _text;
}
```
